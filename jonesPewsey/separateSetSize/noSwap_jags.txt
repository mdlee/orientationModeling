# No-swap memory reproduction, one Jones-Pewsey fit per set size.
# Each run uses a single (kappa, psi). See README.md in this folder.


model {
  pi2 <- 3.141592653589793 * 2
  halfPi <- pi2 / 4

  # Identifiability constraints on psychological locations (semi-circle)
  mu[1] <- 0
  mu[2] ~ dunif(0, halfPi)
  for (j in 3:nStimuli) {
    mu[j] ~ dunif(0, halfPi * 2)
  }

  sigma ~ dunif(0.02, halfPi * 2)
  kappa <- 1 / (sigma * sigma)
  psi ~ dunif(-1, 1)

  for (r in 1:nStimuli) {
    phiMu[r] <- 2 * mu[r]
  }

  for (t in 1:nTrials) {
    yCirc[t] ~ dJonesPewsey(phiMu[s[t]], kappa, psi)
  }
}
