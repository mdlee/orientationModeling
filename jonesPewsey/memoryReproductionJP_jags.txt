# Memory reproduction with swaps (Jones-Pewsey / doubled-angle)
#
# Shared mu across set sizes 3 and 6. Noise (sigma, psi) and swap weights
# (omega3, omega6) are set-size-specific. Likelihoods are two loops so each
# (kappa, psi) pair is a contiguous block of dJonesPewsey nodes.
#
# Requires the Jones-Pewsey JAGS module (./jags-jonesPewsey).
# psi[k] ~ dunif(-1, 1). sigma has a small positive floor so kappa cannot explode.

data {
  omega3priorProbs <- c(1, 1, 1)
  omega6priorProbs <- c(1, 1, 1, 1, 1, 1)
}

model {
  pi2 <- 3.141592653589793 * 2
  halfPi <- pi2 / 4

  # Identifiability constraints on psychological locations (semi-circle)
  mu[1] <- 0
  mu[2] ~ dunif(0, halfPi)
  for (j in 3:nStimuli) {
    mu[j] ~ dunif(0, halfPi * 2)
  }

  for (r in 1:nStimuli) {
    phiMu[r] <- 2 * mu[r]
  }

  sigma[1] ~ dunif(0.02, halfPi * 2)
  sigma[2] ~ dunif(0.02, halfPi * 2)
  psi[1] ~ dunif(-1, 1)
  psi[2] ~ dunif(-1, 1)
  for (k in 1:2) {
    kappa[k] <- 1 / (sigma[k] * sigma[k])
  }

  omega3prior ~ ddirch(omega3priorProbs[])
  omega6prior ~ ddirch(omega6priorProbs[])
  omega3 ~ ddirch(omega3priorProbs[])
  omega6 ~ ddirch(omega6priorProbs[])

  for (t in 1:nTrials3) {
    xi3[t] ~ dcat(omega3[])
    recalled3[t] <- s3[t, xi3[t]]
    yCirc3[t] ~ dJonesPewsey(phiMu[recalled3[t]], kappa[1], psi[1])
  }
  for (t in 1:nTrials6) {
    xi6[t] ~ dcat(omega6[])
    recalled6[t] <- s6[t, xi6[t]]
    yCirc6[t] ~ dJonesPewsey(phiMu[recalled6[t]], kappa[2], psi[2])
  }
}
