# Similarity comparison (wrap-copy Gaussian) — MCMC driver version
# No posterior-predictive copies. Trial latents are sampled but not monitored.

data {
  pi = 3.141592653589793
  wrapOff[1] = 0
  wrapOff[2] = -pi
  wrapOff[3] = pi
}

model {
  mu[1] ~ dunif(0, 1e-6)
  mu[2] ~ dunif(0, pi/2)
  for (j in 3:nStimuli) {
    mu[j] ~ dunif(0, pi)
  }

  sigma ~ dunif(0, pi)

  for (t in 1:nTrials) {
    nu[t] ~ dcat(c(1, 1, 1))

    xA[t] ~ dnorm(mu[a[t]] + wrapOff[nu[t]], 1 / sigma^2)
    xB[t] ~ dnorm(mu[b[t]] + wrapOff[nu[t]], 1 / sigma^2)
    xC[t] ~ dnorm(mu[c[t]] + wrapOff[nu[t]], 1 / sigma^2)
    xD[t] ~ dnorm(mu[d[t]] + wrapOff[nu[t]], 1 / sigma^2)

    dAB[t] = min(abs(xA[t] - xB[t]), pi - abs(xA[t] - xB[t]))
    dCD[t] = min(abs(xC[t] - xD[t]), pi - abs(xC[t] - xD[t]))

    y[t] ~ dinterval(dAB[t] - dCD[t], 0)
  }
}
